  return (
    <div className="space-y-6">
      {/* Header & Batch Selector */}
      <div className="bg-white rounded-2xl shadow-sm border border-gray-100 p-4 mb-6">
        <div className="flex flex-col md:flex-row md:items-center justify-between gap-4">
          <div>
            <h2 className="text-xl font-bold text-gray-900">Ringkasan Tabel Analisis</h2>
            <p className="text-sm text-gray-600 mt-1">
              Ringkasan kecukupan muallimah dan ketersediaan halaqah dalam format tabel
            </p>
          </div>

          <div className="flex items-center gap-3">
            <div className="flex items-center gap-2">
              <Award className="h-4 w-4 text-gray-500" />
              <select
                value={selectedBatchId}
                onChange={(e) => {
                  setSelectedBatchId(e.target.value);
                  setLoading(true);
                }}
                className="px-3 py-2.5 rounded-xl border border-gray-200 text-sm font-bold focus:outline-none focus:ring-2 focus:ring-green-600/20 focus:border-green-600 bg-white cursor-pointer"
              >
                <option value="">-- Pilih Batch --</option>
                {batches.map((b) => (
                  <option key={b.id} value={b.id}>
                    {b.name} {b.status === "active" ? "(Aktif)" : ""}
                  </option>
                ))}
              </select>
            </div>
            <button
              onClick={() => fetchAnalysisData(selectedBatchId)}
              disabled={loading || !selectedBatchId}
              className="px-4 py-2.5 bg-green-600 hover:bg-green-700 text-white rounded-xl font-bold text-sm transition-all shadow-sm hover:shadow-md disabled:opacity-50 flex items-center gap-2"
            >
              {loading ? (
                <div className="animate-spin rounded-full h-4 w-4 border-2 border-white/20 border-t-white" />
              ) : (
                <BarChart3 className="h-4 w-4" />
              )}
              Refresh
            </button>
          </div>
        </div>
      </div>

      {/* Mode Selectors */}
      <div className="flex flex-col gap-4">
        {/* Program Filter */}
        <div className="flex flex-wrap gap-2 p-1.5 bg-white border border-gray-100 rounded-xl shadow-sm self-start">
          {[
            { id: "semua", label: "Semua Program" },
            { id: "tikrar", label: "Tikrar Tahfidz" },
            { id: "pra_tikrar", label: "Pra Tikrar" },
            { id: "kelas_berbayar", label: "Kelas Berbayar" },
          ].map((tab) => (
            <button
              key={tab.id}
              onClick={() => {
                setProgramTab(tab.id as any);
                setLoading(true);
              }}
              className={`px-4 py-2 rounded-lg text-sm font-bold transition-all ${
                programTab === tab.id
                  ? "bg-blue-50 text-blue-700 border border-blue-100 shadow-sm"
                  : "text-gray-500 hover:bg-gray-50 hover:text-gray-700 border border-transparent"
              }`}
            >
              {tab.label}
            </button>
          ))}
        </div>

        {/* Data Source Filter */}
        <div className="flex flex-wrap gap-2 p-1.5 bg-white border border-gray-100 rounded-xl shadow-sm self-start">
          {[
            { id: "pendaftar", label: "Berdasarkan Pendaftar (Estimasi)" },
            { id: "daftar_ulang", label: "Berdasarkan Daftar Ulang (Aktual)" },
          ].map((mode) => (
            <button
              key={mode.id}
              onClick={() => {
                setAnalysisMode(mode.id as any);
                setLoading(true);
              }}
              className={`px-4 py-2 rounded-lg text-sm font-bold transition-all ${
                analysisMode === mode.id
                  ? "bg-green-50 text-green-700 border border-green-100 shadow-sm"
                  : "text-gray-500 hover:bg-gray-50 hover:text-gray-700 border border-transparent"
              }`}
            >
              {mode.label}
            </button>
          ))}
        </div>
      </div>

      {/* Main Table */}
      {halaqahData.length > 0 ? (
        <div className="bg-white rounded-2xl shadow-sm border border-gray-200 overflow-hidden">
          <div className="overflow-x-auto">
            <table className="w-full text-left text-sm border-collapse">
              <thead className="bg-gray-50/80 border-b border-gray-200">
                <tr>
                  <th className="px-6 py-4 font-bold text-gray-700 whitespace-nowrap">Juz</th>
                  <th className="px-6 py-4 font-bold text-gray-700 text-center whitespace-nowrap">Thalibah</th>
                  <th className="px-6 py-4 font-bold text-gray-700 text-center whitespace-nowrap">Halaqah</th>
                  <th className="px-6 py-4 font-bold text-gray-700 text-center whitespace-nowrap">Kapasitas</th>
                  <th className="px-6 py-4 font-bold text-gray-700 text-center whitespace-nowrap">Terisi</th>
                  <th className="px-6 py-4 font-bold text-gray-700 text-center whitespace-nowrap">Tersedia</th>
                  <th className="px-6 py-4 font-bold text-gray-700 whitespace-nowrap">Detail Muallimah</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-gray-100">
                {halaqahData.map((juz: any) => {
                  const isOvercapacity = juz.total_thalibah > juz.total_capacity;

                  return (
                    <tr key={juz.juz_number} className="hover:bg-gray-50/50 transition-colors align-top">
                      <td className="px-6 py-4">
                        <div className="font-bold text-lg text-gray-900">{juz.juz_name}</div>
                        {isOvercapacity && (
                          <span className="inline-flex items-center gap-1 text-[10px] font-bold text-red-600 bg-red-50 px-2 py-0.5 rounded-full mt-2 border border-red-100 whitespace-nowrap">
                            <AlertTriangle className="w-3 h-3" />
                            Butuh {juz.needed_halaqah} Muallimah
                          </span>
                        )}
                      </td>
                      <td className="px-6 py-4 text-center align-middle">
                        <span className="font-bold text-base text-gray-900">{juz.total_thalibah}</span>
                      </td>
                      <td className="px-6 py-4 text-center align-middle">
                        <span className="font-bold text-base text-gray-900">{juz.total_halaqah}</span>
                      </td>
                      <td className="px-6 py-4 text-center align-middle">
                        <span className="font-bold text-base text-gray-900">{juz.total_capacity}</span>
                      </td>
                      <td className="px-6 py-4 text-center align-middle">
                        <span className="font-bold text-base text-blue-700 bg-blue-50 px-2.5 py-1 rounded-lg border border-blue-100">
                          {juz.total_filled}
                        </span>
                      </td>
                      <td className="px-6 py-4 text-center align-middle">
                        <span className={`font-bold text-base px-2.5 py-1 rounded-lg border ${isOvercapacity ? 'text-red-700 bg-red-50 border-red-100' : 'text-green-700 bg-green-50 border-green-100'}`}>
                          {juz.total_available}
                        </span>
                      </td>
                      <td className="px-6 py-4 min-w-[320px]">
                        {juz.halaqah_details?.length > 0 ? (
                          <div className="space-y-3">
                            {juz.halaqah_details.map((h: any, idx: number) => (
                              <div key={idx} className="bg-white border border-gray-100 rounded-lg p-3 shadow-sm">
                                <div className="font-bold text-sm text-gray-900 flex justify-between items-start gap-2 mb-1.5">
                                  <span>{h.muallimah_name}</span>
                                  {h.class_type && (
                                    <span className="text-[10px] px-1.5 py-0.5 rounded bg-gray-100 text-gray-600 border border-gray-200 whitespace-nowrap font-medium">
                                      {h.class_type.split(',').map((t: string) => t.trim().replace(/_/g, ' ')).join(', ')}
                                    </span>
                                  )}
                                </div>
                                <div className="flex flex-col gap-1.5 text-[11px] text-gray-600">
                                  <div className="flex justify-between items-center bg-gray-50 px-2 py-1 rounded">
                                    <span className="font-medium">Kapasitas/Terisi:</span>
                                    <span className="font-bold">
                                      {h.current_students} / {h.max_students}
                                    </span>
                                  </div>
                                  
                                  {h.schedules?.length > 0 ? (
                                    <div className="space-y-1">
                                      {h.schedules.map((s: any, sIdx: number) => (
                                        <div key={sIdx} className={`flex justify-between items-center ${s.is_allocated_here ? 'text-green-700 font-semibold' : 'text-gray-500'}`}>
                                          <span>{s.type} {s.is_backup ? "(Cadangan)" : ""} :</span>
                                          <span>{s.day_name} {s.start_time !== "-" ? `(${s.start_time}-${s.end_time})` : ""}</span>
                                        </div>
                                      ))}
                                    </div>
                                  ) : (
                                    <div className="text-gray-400 italic">Jadwal belum tersedia</div>
                                  )}
                                </div>
                              </div>
                            ))}
                          </div>
                        ) : (
                          <div className="text-gray-400 italic text-sm py-2">Belum ada muallimah.</div>
                        )}
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          </div>
        </div>
      ) : (
        <div className="text-center py-16 bg-white rounded-2xl border border-gray-200 border-dashed">
          <BookOpen className="h-12 w-12 text-gray-300 mx-auto mb-4" />
          <h3 className="text-lg font-bold text-gray-900 mb-1">
            Data Analisis Kosong
          </h3>
          <p className="text-gray-500 text-sm max-w-sm mx-auto">
            Belum ada data halaqah atau pendaftar yang dapat dianalisis untuk batch dan program ini.
          </p>
        </div>
      )}
    </div>
  );
}
